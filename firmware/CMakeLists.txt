cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH})
    message(FATAL_ERROR "Defina PICO_SDK_PATH para um checkout do pico-sdk com submodulos.")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(infineon_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(infineon_pico main.c)
target_link_libraries(infineon_pico pico_stdlib hardware_i2c)
pico_enable_stdio_usb(infineon_pico 1)
pico_enable_stdio_uart(infineon_pico 0)
pico_add_extra_outputs(infineon_pico)
